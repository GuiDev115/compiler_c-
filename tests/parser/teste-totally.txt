// Função com todos os operadores
float calcula(float x, float y) {
    float resultado;
    
    // Todos os operadores relacionais
    if (x <= y) resultado = x + y;
    if (x < y) resultado = x - y;
    if (x > y) resultado = x * y;
    if (x >= y) resultado = x / y;
    if (x == y) resultado = x;
    if (x != y) resultado = y;
    
    return resultado;
}
